cmake_minimum_required(VERSION 3.20)
project(husk_arm64_exec LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(DYNARMIC_FRONTENDS A64 CACHE STRING "" FORCE)
set(DYNARMIC_TESTS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_USE_BUNDLED_EXTERNALS ON CACHE BOOL "" FORCE)
set(DYNARMIC_WARNINGS_AS_ERRORS OFF CACHE BOOL "" FORCE)
set(DYNARMIC_ENABLE_NO_EXECUTE_SUPPORT ON CACHE BOOL "" FORCE)
add_subdirectory("${DYNARMIC_SOURCE_DIR}" dynarmic)
set(DIRECT_SOURCES husk-arm64-direct.cpp husk-arm64-direct-entry.S husk-arm64-fastmem.cpp husk-arm64-bulk.cpp)
add_library(husk_arm64_exec STATIC husk-arm64-exec.cpp ${DIRECT_SOURCES})
target_include_directories(husk_arm64_exec PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(husk_arm64_exec PUBLIC dynarmic PRIVATE oaknut)

enable_testing()
add_executable(arm64_bulk_test ../../tests/arm64-exec/bulk.cpp)
target_link_libraries(arm64_bulk_test PRIVATE husk_arm64_exec)
add_test(NAME arm64_bulk COMMAND arm64_bulk_test)
add_executable(arm64_exec_test ../../tests/arm64-exec/engine.cpp)
target_link_libraries(arm64_exec_test PRIVATE husk_arm64_exec)
add_test(NAME arm64_exec COMMAND arm64_exec_test)
add_test(NAME arm64_direct COMMAND arm64_exec_test)
set_tests_properties(arm64_direct PROPERTIES ENVIRONMENT "HUSK_ARM64_BACKEND=direct")
add_test(NAME arm64_direct_checked COMMAND arm64_exec_test)
set_tests_properties(arm64_direct_checked PROPERTIES ENVIRONMENT "HUSK_ARM64_BACKEND=direct;HUSK_ARM64_LOOP_MAP=0")
add_executable(arm64_direct_differential ../../tests/arm64-exec/direct-differential.cpp)
target_link_libraries(arm64_direct_differential PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_differential COMMAND arm64_direct_differential)
add_executable(arm64_direct_simd ../../tests/arm64-exec/direct-simd.cpp)
target_link_libraries(arm64_direct_simd PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_simd COMMAND arm64_direct_simd)
add_executable(arm64_direct_slices ../../tests/arm64-exec/direct-slices.cpp)
target_link_libraries(arm64_direct_slices PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_slices COMMAND arm64_direct_slices)
set_tests_properties(arm64_direct_slices PROPERTIES TIMEOUT 30)
add_executable(arm64_direct_atomics ../../tests/arm64-exec/direct-atomics.cpp)
target_link_libraries(arm64_direct_atomics PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_atomics COMMAND arm64_direct_atomics)
add_executable(arm64_direct_fastmem ../../tests/arm64-exec/direct-fastmem.cpp)
target_link_libraries(arm64_direct_fastmem PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_fastmem COMMAND arm64_direct_fastmem)
set_tests_properties(arm64_direct_fastmem PROPERTIES TIMEOUT 30)
add_executable(arm64_direct_cache ../../tests/arm64-exec/direct-cache.cpp)
target_link_libraries(arm64_direct_cache PRIVATE husk_arm64_exec)
add_test(NAME arm64_direct_cache COMMAND arm64_direct_cache)
set_tests_properties(arm64_direct_cache PROPERTIES TIMEOUT 30)
foreach(suite differential simd atomics)
    add_test(NAME arm64_fastmem_${suite} COMMAND arm64_direct_${suite})
    set_tests_properties(arm64_fastmem_${suite} PROPERTIES
        ENVIRONMENT "HUSK_ARM64_FASTMEM=1;HUSK_ARM64_LOOP_MAP=0" TIMEOUT 30)
endforeach()

# Exercise the same C ABI wrapper shipped in the iOS archive on the host.
add_executable(arm64_direct_api_test ../../tests/arm64-exec/engine.cpp
    husk-arm64-direct-api.cpp ${DIRECT_SOURCES})
target_include_directories(arm64_direct_api_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(arm64_direct_api_test PRIVATE oaknut)
add_test(NAME arm64_direct_api COMMAND arm64_direct_api_test)
set_tests_properties(arm64_direct_api PROPERTIES ENVIRONMENT "HUSK_ARM64_BACKEND=direct")
